kubectl proxy

اجرای یک پراکسی به سرور API کوبرنتیز

خلاصه

این دستور یک سرور پراکسی یا دروازه در سطح برنامه (application-level gateway) میان localhost و سرور API کوبرنتیز ایجاد می‌کند. همچنین می‌توان با آن محتوای ایستا را روی یک مسیر HTTP مشخص ارائه کرد. همه‌ی داده‌های ورودی از یک پورت وارد می‌شوند و به پورت سرور API کوبرنتیز راه دور هدایت می‌شوند؛ تنها استثنا مسیری است که با مسیر محتوای ایستا مطابقت داشته باشد.

kubectl proxy [--port=PORT] [--www=static-dir] [--www-prefix=prefix] [--api-prefix=prefix]

مثال‌ها

  # To proxy all of the Kubernetes API and nothing else
  kubectl proxy --api-prefix=/
  
  # To proxy only part of the Kubernetes API and also some static files
  # You can get pods info with 'curl localhost:8001/api/v1/pods'
  kubectl proxy --www=/my/files --www-prefix=/static/ --api-prefix=/api/
  
  # To proxy the entire Kubernetes API at a different root
  # You can get pods info with 'curl localhost:8001/custom/api/v1/pods'
  kubectl proxy --api-prefix=/custom/
  
  # Run a proxy to the Kubernetes API server on port 8011, serving static content from ./local/www/
  kubectl proxy --port=8011 --www=./local/www/
  
  # Run a proxy to the Kubernetes API server on an arbitrary local port
  # The chosen port for the server will be output to stdout
  kubectl proxy --port=0
  
  # Run a proxy to the Kubernetes API server, changing the API prefix to k8s-api
  # This makes e.g. the pods API available at localhost:8001/k8s-api/v1/pods/
  kubectl proxy --api-prefix=/k8s-api

گزینه‌ها

--accept-hosts string     Default: "^localhost$,^127\.0\.0\.1$,^\[::1\]$"

عبارت باقاعده (regular expression) برای میزبان‌هایی که پراکسی باید آن‌ها را بپذیرد.
--accept-paths string     Default: "^.*"

عبارت باقاعده برای مسیرهایی که پراکسی باید آن‌ها را بپذیرد.
--address string     Default: "127.0.0.1"

نشانی IP که سرویس‌دهی روی آن انجام می‌شود.
--api-prefix string     Default: "/"

پیشوندی که API پراکسی‌شده زیر آن ارائه می‌شود.
--append-server-path

اگر true باشد، مسیر سرورِ کانتکست kube به‌طور خودکار به انتهای هر درخواست افزوده می‌شود.
--disable-filter

اگر true باشد، فیلتر کردن درخواست‌ها در پراکسی غیرفعال می‌شود. این کار خطرناک است و اگر همراه با یک پورت در دسترس استفاده شود، ممکن است شما را در برابر حملات XSRF آسیب‌پذیر کند.
-h, --help

راهنمای دستور proxy
--keepalive duration

فیلد keepalive دوره‌ی keep-alive را برای یک اتصال شبکه‌ی فعال تعیین می‌کند. برای غیرفعال کردن keepalive مقدار آن را 0 قرار دهید.
-p, --port int     Default: 8001

پورتی که پراکسی روی آن اجرا می‌شود. برای انتخاب یک پورت تصادفی، مقدار آن را 0 قرار دهید.
--reject-methods string     Default: "^$"

عبارت باقاعده برای متدهای HTTP که پراکسی باید آن‌ها را رد کند (برای نمونه --reject-methods='POST,PUT,PATCH').
--reject-paths string     Default: "^/api/.*/pods/.*/exec,
^/api/.*/pods/.*/attach"

عبارت باقاعده برای مسیرهایی که پراکسی باید آن‌ها را رد کند. مسیرهایی که در اینجا مشخص می‌شوند، حتی اگر با --accept-paths پذیرفته شده باشند، رد خواهند شد.
-u, --unix-socket string

سوکت یونیکسی که پراکسی روی آن اجرا می‌شود.
-w, --www string

علاوه بر API، فایل‌های ایستا را نیز از دایرکتوری داده‌شده و زیر پیشوند مشخص‌شده ارائه می‌کند.
-P, --www-prefix string     Default: "/static/"

پیشوندی که فایل‌های ایستا زیر آن ارائه می‌شوند، در صورتی که دایرکتوری فایل‌های ایستا مشخص شده باشد.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که عملیات به جای او (impersonate) انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات به جای آن انجام می‌شود. برای مشخص کردن چند گروه می‌توان این فلگ را تکرار کرد.
--as-uid string

شناسه‌ی UID که عملیات به جای آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات به جای آن انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توان این فلگ را تکرار کرد.
--cache-dir string     Default: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام کانتکستی از kubeconfig که باید استفاده شود
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌های ارسالی به سرور غیرفعال می‌شود
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نمی‌شود. این کار اتصال‌های HTTPS شما را ناامن می‌کند
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های CLI استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات (preferences) استفاده می‌شود. می‌توان این قابلیت را با export کردن feature gate به صورت KUBECTL_KUBERC=false یا با خاموش کردن قابلیت KUBERC=off غیرفعال کرد.
--match-server-version

الزام به این‌که نسخه‌ی سرور با نسخه‌ی کلاینت مطابقت داشته باشد
-n, --namespace string

در صورت وجود، دامنه‌ی فضای نام برای این درخواست CLI را تعیین می‌کند
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در سرور API
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود. یکی از مقادیر (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی که برای درخواست‌های ارسالی به سرور API استفاده می‌شود
--request-timeout string     Default: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست منفرد به سرور، منتظر می‌ماند. مقادیر غیرصفر باید یک واحد زمانی متناظر داشته باشند (برای نمونه 1s، 2m، 3h). مقدار صفر به این معناست که برای درخواست‌ها مهلت زمانی (timeout) در نظر گرفته نمی‌شود.
-s, --server string

نشانی و پورت سرور API کوبرنتیز
--storage-driver-buffer-duration duration     Default: 1m0s

عملیات نوشتن در درایور ذخیره‌سازی به مدت این بازه‌ی زمانی در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در بک‌اندهای غیرحافظه‌ای (non memory backends) ثبت می‌شوند
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده
--storage-driver-host string     Default: "localhost:8086"

میزبان و پورت پایگاه داده به صورت host:port
--storage-driver-password string     Default: "root"

گذرواژه‌ی پایگاه داده
--storage-driver-secure

استفاده از اتصال امن با پایگاه داده
--storage-driver-table string     Default: "stats"

نام جدول
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر مشخص نشود، از نام میزبانی که برای برقراری ارتباط با سرور به کار رفته استفاده می‌شود
--token string

توکن حامل (bearer token) برای احراز هویت در سرور API
--user string

نام کاربری از kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در سرور API
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینه‌ی --version=vX.Y.Z... نسخه‌ی گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافت‌شده از سرور را به‌عنوان خطا در نظر می‌گیرد و با یک کد خروج غیرصفر خارج می‌شود
