kubectl annotate

به‌روزرسانی حاشیه‌نویسی‌های (annotation) یک منبع

خلاصه

حاشیه‌نویسی‌های (annotation) یک یا چند منبع را به‌روزرسانی می‌کند.

همه آبجکت‌های کوبرنتیز قابلیت ذخیره داده‌های اضافی را به‌صورت حاشیه‌نویسی (annotation) همراه با آبجکت دارند. حاشیه‌نویسی‌ها جفت‌های کلید/مقدار هستند که می‌توانند از برچسب‌ها بزرگ‌تر باشند و مقادیر رشته‌ای دلخواه، مانند JSON ساختاریافته، را در بر بگیرند. ابزارها و افزونه‌های سیستمی می‌توانند از حاشیه‌نویسی‌ها برای ذخیره داده‌های خود استفاده کنند.

تلاش برای تنظیم حاشیه‌نویسی‌ای که از قبل وجود دارد با شکست مواجه می‌شود، مگر اینکه --overwrite تنظیم شده باشد. اگر --resource-version مشخص شده باشد و با نسخه فعلی منبع روی سرور مطابقت نداشته باشد، دستور با شکست مواجه خواهد شد.

برای مشاهده فهرست کامل منابع پشتیبانی‌شده از "kubectl api-resources" استفاده کنید.

kubectl annotate [--overwrite] (-f FILENAME | TYPE NAME) KEY_1=VAL_1 ... KEY_N=VAL_N [--resource-version=version]

مثال‌ها

  # Update pod 'foo' with the annotation 'description' and the value 'my frontend'
  # If the same annotation is set multiple times, only the last value will be applied
  kubectl annotate pods foo description='my frontend'
  
  # Update a pod identified by type and name in "pod.json"
  kubectl annotate -f pod.json description='my frontend'
  
  # Update pod 'foo' with the annotation 'description' and the value 'my frontend running nginx', overwriting any existing value
  kubectl annotate --overwrite pods foo description='my frontend running nginx'
  
  # Update all pods in the namespace
  kubectl annotate pods --all description='my frontend running nginx'
  
  # Update pod 'foo' only if the resource is unchanged from version 1
  kubectl annotate pods foo description='my frontend running nginx' --resource-version=1
  
  # Update pod 'foo' by removing an annotation named 'description' if it exists
  # Does not require the --overwrite flag
  kubectl annotate pods foo description-

گزینه‌ها

--all

همه منابع موجود در فضای نام (namespace) برای نوع‌های منبع مشخص‌شده را انتخاب می‌کند.
-A, --all-namespaces

در صورت وجود، آبجکت(های) درخواستی را در همه فضاهای نام (namespace) فهرست می‌کند. فضای نام موجود در context فعلی نادیده گرفته می‌شود، حتی اگر با --namespace مشخص شده باشد.
--allow-missing-template-keys     Default: true

اگر true باشد، هرگونه خطا در قالب‌ها (template) را هنگامی که یک فیلد یا کلید map در قالب وجود ندارد نادیده می‌گیرد. فقط برای قالب‌های خروجی golang و jsonpath اعمال می‌شود.
--dry-run string[="unchanged"]     Default: "none"

باید یکی از مقادیر "none"، "server" یا "client" باشد. در راهبرد client، فقط آبجکتی که ارسال می‌شد چاپ می‌شود، بدون آنکه ارسال شود. در راهبرد server، درخواست سمت سرور ثبت می‌شود، بدون آنکه منبع ذخیره (persist) شود.
--field-manager string     Default: "kubectl-annotate"

نام مدیری که برای ردیابی مالکیت فیلدها استفاده می‌شود.
--field-selector string

انتخابگر (selector) (پرس‌وجوی فیلد) برای فیلتر کردن؛ از '='، '==' و '!=' پشتیبانی می‌کند. (مثال: --field-selector key1=value1,key2=value2). سرور برای هر نوع فقط تعداد محدودی پرس‌وجوی فیلد را پشتیبانی می‌کند.
-f, --filename strings

برای شناسایی منبع.
-h, --help

راهنما برای annotate
-k, --kustomize string

یک دایرکتوری kustomization را پردازش می‌کند. این فلگ را نمی‌توان همراه با -f یا -R استفاده کرد.
--list

اگر true باشد، حاشیه‌نویسی‌های (annotation) یک منبع مشخص را نمایش می‌دهد.
--local

اگر true باشد، annotate با api-server ارتباط برقرار نمی‌کند و به‌صورت محلی اجرا می‌شود.
-o, --output string

قالب خروجی. یکی از: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
--overwrite

اگر true باشد، اجازه می‌دهد حاشیه‌نویسی‌ها بازنویسی شوند؛ در غیر این صورت به‌روزرسانی‌هایی که حاشیه‌نویسی‌های موجود را بازنویسی می‌کنند رد می‌شوند.
-R, --recursive     Default: true

دایرکتوری استفاده‌شده در -f, --filename را به‌صورت بازگشتی پردازش می‌کند. زمانی مفید است که بخواهید مانیفست‌های مرتبطی را که در یک دایرکتوری قرار دارند مدیریت کنید.
--resource-version string

اگر خالی نباشد، به‌روزرسانی حاشیه‌نویسی (annotation) تنها در صورتی موفق می‌شود که این مقدار همان resource-version فعلی آبجکت باشد. فقط هنگام مشخص کردن یک منبع واحد معتبر است.
-l, --selector string

انتخابگر (selector) (پرس‌وجوی برچسب) برای فیلتر کردن؛ از '='، '==' و '!=' پشتیبانی می‌کند. (مثال: -l key1=value1,key2=value2)
--show-managed-fields

اگر true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، managedFields را نگه می‌دارد.
--template string

رشته قالب یا مسیر فایل قالب برای استفاده هنگام -o=go-template و -o=go-template-file. قالب‌بندی آن از نوع قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که برای عملیات جعل هویت (impersonate) می‌شود. کاربر می‌تواند یک کاربر عادی یا یک service account در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که برای عملیات جعل هویت (impersonate) می‌شود؛ این فلگ را می‌توان برای مشخص کردن چند گروه تکرار کرد.
--as-uid string

UID که برای عملیات جعل هویت (impersonate) می‌شود.
--as-user-extra strings

اطلاعات اضافی کاربر که برای عملیات جعل هویت (impersonate) می‌شود؛ این فلگ را می‌توان برای مشخص کردن چند مقدار برای یک کلید یکسان تکرار کرد.
--cache-dir string     Default: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض کش
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستر kubeconfig که باید استفاده شود
--context string

نام context در kubeconfig که باید استفاده شود
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همه درخواست‌ها به سرور غیرفعال می‌شود
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نخواهد شد. این کار اتصال‌های HTTPS شما را ناامن می‌کند
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های CLI استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای تنظیمات ترجیحی استفاده می‌شود. این قابلیت را می‌توان با export کردن feature gate به‌صورت KUBECTL_KUBERC=false یا با خاموش کردن قابلیت از طریق KUBERC=off غیرفعال کرد.
--match-server-version

نسخه سرور را ملزم به تطابق با نسخه کلاینت می‌کند
-n, --namespace string

در صورت وجود، دامنه فضای نام (namespace) برای این درخواست CLI
--password string

گذرواژه برای احراز هویت پایه (basic authentication) به API server
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود. یکی از (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پروکسی که برای درخواست‌ها به API server استفاده می‌شود
--request-timeout string     Default: "0"

مدت زمان انتظار قبل از صرف‌نظر کردن از یک درخواست واحد به سرور. مقادیر غیرصفر باید شامل واحد زمانی مربوطه باشند (مثلاً 1s، 2m، 3h). مقدار صفر به معنای عدم اعمال زمان انتظار (timeout) برای درخواست‌ها است.
-s, --server string

نشانی و درگاه (port) سرور API کوبرنتیز
--storage-driver-buffer-duration duration     Default: 1m0s

نوشتن‌ها در storage driver به مدت این بازه بافر می‌شوند و به‌صورت یک تراکنش واحد در بک‌اندهای غیرحافظه‌ای ثبت (commit) می‌شوند
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده
--storage-driver-host string     Default: "localhost:8086"

میزبان:درگاه پایگاه داده (host:port)
--storage-driver-password string     Default: "root"

گذرواژه پایگاه داده
--storage-driver-secure

استفاده از اتصال امن با پایگاه داده
--storage-driver-table string     Default: "stats"

نام جدول
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر ارائه نشود، از نام میزبانی که برای ارتباط با سرور استفاده شده است استفاده می‌شود
--token string

توکن Bearer برای احراز هویت به API server
--user string

نام کاربر kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه (basic authentication) به API server
--version version[=true]

--version و --version=raw اطلاعات نسخه را چاپ کرده و خارج می‌شوند؛ --version=vX.Y.Z... نسخه گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافتی از سرور را به‌عنوان خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود
