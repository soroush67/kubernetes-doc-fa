kubectl wait

منتظر ماندن برای برقرار شدن شرطی مشخص روی یک یا چند منبع

خلاصه

منتظر می‌ماند تا شرطی مشخص روی یک یا چند منبع برقرار شود.

این فرمان چند منبع را دریافت می‌کند و منتظر می‌ماند تا شرط تعیین‌شده در فیلد Status همهٔ منابع داده‌شده دیده شود.

همچنین اگر کلیدواژهٔ "create" یا "delete" را به‌عنوان مقدار پرچم ‎--for‎ بدهید، این فرمان منتظر می‌ماند تا مجموعهٔ منابع داده‌شده ایجاد یا حذف شوند.

هنگامی که شرط تعیین‌شده برقرار شود، پیامی مبنی بر موفقیت در stdout چاپ می‌شود. برای تغییر مقصد خروجی می‌توانید از گزینهٔ ‎-o‎ استفاده کنید.

kubectl wait ([-f FILENAME] | resource.group/resource.name | resource.group [(-l label | --all)]) [--for=create|--for=delete|--for condition=available|--for=jsonpath='{}'[=value]]

مثال‌ها

  # Wait for the pod "busybox1" to contain the status condition of type "Ready"
  kubectl wait --for=condition=Ready pod/busybox1
  
  # The default value of status condition is true; you can wait for other targets after an equal delimiter (compared after Unicode simple case folding, which is a more general form of case-insensitivity)
  kubectl wait --for=condition=Ready=false pod/busybox1
  
  # Wait for the pod "busybox1" to contain the status phase to be "Running"
  kubectl wait --for=jsonpath='{.status.phase}'=Running pod/busybox1
  
  # Wait for pod "busybox1" to be Ready
  kubectl wait --for='jsonpath={.status.conditions[?(@.type=="Ready")].status}=True' pod/busybox1
  
  # Wait for the service "loadbalancer" to have ingress
  kubectl wait --for=jsonpath='{.status.loadBalancer.ingress}' service/loadbalancer
  
  # Wait for the secret "busybox1" to be created, with a timeout of 30s
  kubectl create secret generic busybox1
  kubectl wait --for=create secret/busybox1 --timeout=30s
  
  # Wait for the pod "busybox1" to be deleted, with a timeout of 60s, after having issued the "delete" command
  kubectl delete pod/busybox1
  kubectl wait --for=delete pod/busybox1 --timeout=60s
  
  # Wait for pod "busybox1" to be created AND reach the "Ready" status condition
  kubectl wait --for=condition=Ready --for=create pod/busybox1
  
  # Wait for pod "busybox1" to reach the "Ready" status OR for its containers to report a "False" readiness state
  until kubectl wait pod/busybox1 --for=condition=Ready --timeout=1s 2>/dev/null || \
  kubectl wait pod/busybox1 --for=condition=ContainersReady=False --timeout=1s 2>/dev/null; \
  do echo "Checking conditions..."; sleep 1; done

گزینه‌ها

--all

همهٔ منابعِ انواع منبع تعیین‌شده را در فضای نام (namespace) انتخاب می‌کند.
-A, --all-namespaces

در صورت وجود، آبجکت‌های درخواست‌شده را در همهٔ فضاهای نام فهرست می‌کند. فضای نامِ زمینهٔ (context) فعلی نادیده گرفته می‌شود، حتی اگر با ‎--namespace‎ تعیین شده باشد.
--allow-missing-template-keys     Default: true

اگر true باشد، هنگامی که فیلد یا کلید نگاشتی (map) در قالب وجود نداشته باشد، از هر خطایی در قالب‌ها چشم‌پوشی می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath کاربرد دارد.
--field-selector string

انتخابگر (selector) مبتنی بر فیلد (پرس‌وجوی فیلد) برای فیلتر کردن، که از '='، '==' و '!=' پشتیبانی می‌کند (برای مثال ‎--field-selector key1=value1,key2=value2‎). سرور برای هر نوع فقط از تعداد محدودی پرس‌وجوی فیلد پشتیبانی می‌کند.
-f, --filename strings

برای شناسایی منبع.
--for strings

شرطی که باید منتظر آن ماند: [create|delete|condition=condition-name[=condition-value]|jsonpath='{JSONPath expression}'=[JSONPath value]]. مقدار پیش‌فرض condition-value برابر true است. مقادیر شرط پس از تبدیل سادهٔ حروف یونیکد (Unicode simple case folding) مقایسه می‌شوند که شکل کلی‌ترِ حساس نبودن به بزرگی و کوچکی حروف است. چند شرط پشتیبانی می‌شود و این شرط‌ها به ترتیب با یکدیگر AND می‌شوند. اگر ‎--for=create‎ داده شود، همیشه ابتدا منتظر آن می‌ماند.
-h, --help

راهنمای wait
--local

اگر true باشد، حاشیه‌نویسی (annotation) با api-server ارتباط برقرار نمی‌کند و به‌صورت محلی اجرا می‌شود.
-o, --output string

قالب خروجی. یکی از این مقادیر: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
-R, --recursive     Default: true

پوشهٔ استفاده‌شده در ‎-f, --filename‎ را به‌صورت بازگشتی پردازش می‌کند. این گزینه زمانی مفید است که بخواهید مانیفست‌های مرتبطی را که در یک پوشه سازمان‌دهی شده‌اند مدیریت کنید.
-l, --selector string

انتخابگر مبتنی بر برچسب (پرس‌وجوی برچسب) برای فیلتر کردن، که از '='، '==' و '!=' پشتیبانی می‌کند (برای مثال ‎-l key1=value1,key2=value2‎).
--show-managed-fields

اگر true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشتهٔ قالب یا مسیر فایل قالب که هنگام استفاده از ‎-o=go-template‎ یا ‎-o=go-template-file‎ به کار می‌رود. قالب‌بندی آن مطابق قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].
--timeout duration     Default: 30s

مدت زمان انتظار پیش از منصرف شدن. مقدار صفر یعنی یک بار بررسی شود و منتظر نماند، و مقدار منفی یعنی به مدت یک هفته منتظر بماند.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری‌ای که عملیات از طرف آن جعل هویت (impersonate) می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام باشد.
--as-group strings

گروهی که عملیات از طرف آن جعل هویت می‌شود. برای تعیین چند گروه می‌توان این پرچم را تکرار کرد.
--as-uid string

شناسهٔ UID که عملیات از طرف آن جعل هویت می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات با آن جعل هویت می‌شود. برای تعیین چند مقدار برای یک کلید می‌توان این پرچم را تکرار کرد.
--cache-dir string     Default: "$HOME/.kube/cache"

پوشهٔ پیش‌فرض حافظهٔ نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستر kubeconfig مورد نظر برای استفاده
--context string

نام زمینهٔ kubeconfig مورد نظر برای استفاده
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همهٔ درخواست‌ها به سرور غیرفعال می‌شود.
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نمی‌شود. این کار اتصال‌های HTTPS شما را ناامن می‌کند.
--kubeconfig string

مسیر فایل kubeconfig برای استفاده در درخواست‌های CLI.
--kuberc string

مسیر فایل kuberc برای استفاده در تنظیمات ترجیحی. می‌توان آن را با export کردن feature gate ‏KUBECTL_KUBERC=false یا خاموش کردن قابلیت با KUBERC=off غیرفعال کرد.
--match-server-version

الزام به یکسان بودن نسخهٔ سرور با نسخهٔ کلاینت
-n, --namespace string

در صورت وجود، دامنهٔ فضای نام برای این درخواست CLI
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در API server
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود. یکی از این مقادیر: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی برای استفاده در درخواست‌ها به API server
--request-timeout string     Default: "0"

مدت زمان انتظار پیش از منصرف شدن از یک درخواست منفرد به سرور. مقادیر غیرصفر باید واحد زمانی متناظر داشته باشند (برای مثال 1s، 2m، 3h). مقدار صفر یعنی درخواست‌ها هیچ‌گاه منقضی نشوند.
-s, --server string

نشانی و درگاه (port) سرور API کوبرنتیز
--storage-driver-buffer-duration duration     Default: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به این مدت بافر می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های (backend) غیرحافظه‌ای ثبت می‌شوند.
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده
--storage-driver-host string     Default: "localhost:8086"

میزبان و درگاه پایگاه داده به‌صورت host:port
--storage-driver-password string     Default: "root"

گذرواژهٔ پایگاه داده
--storage-driver-secure

استفاده از اتصال امن به پایگاه داده
--storage-driver-table string     Default: "stats"

نام جدول
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سرور برای اعتبارسنجی گواهی سرور. اگر تعیین نشود، نام میزبانی که برای ارتباط با سرور به کار رفته است استفاده می‌شود.
--token string

توکن Bearer برای احراز هویت در API server
--user string

نام کاربر kubeconfig مورد نظر برای استفاده
--username string

نام کاربری برای احراز هویت پایه در API server
--version version[=true]

گزینه‌های ‎--version‎ و ‎--version=raw‎ اطلاعات نسخه را چاپ می‌کنند و خارج می‌شوند؛ ‎--version=vX.Y.Z...‎ نسخهٔ گزارش‌شده را تنظیم می‌کند.
--warnings-as-errors

هشدارهای دریافتی از سرور را خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود.
