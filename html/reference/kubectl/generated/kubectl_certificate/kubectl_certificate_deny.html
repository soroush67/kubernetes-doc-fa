kubectl certificate deny

رد کردن یک درخواست امضای گواهی

خلاصه

یک درخواست امضای گواهی را رد می‌کند.

فرمان kubectl certificate deny به مدیر کلاستر امکان می‌دهد یک درخواست امضای گواهی (Certificate Signing Request یا CSR) را رد کند. این عمل به کنترل‌کننده‌ی امضای گواهی اعلام می‌کند که برای درخواست‌کننده گواهی صادر نکند.

kubectl certificate deny (-f FILENAME | NAME)

مثال‌ها

  # Deny CSR 'csr-sqgzp'
  kubectl certificate deny csr-sqgzp

گزینه‌ها

--allow-missing-template-keys     پیش‌فرض: true

اگر مقدار آن true باشد، هرگاه یک فیلد یا کلید نگاشت (map) در قالب وجود نداشته باشد، از همه‌ی خطاهای مربوط به قالب چشم‌پوشی می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath اعمال می‌شود.
-f, --filename strings

نام فایل، دایرکتوری یا URL فایل‌هایی که منبعِ موردنظر برای به‌روزرسانی را مشخص می‌کنند.
--force

درخواست CSR را به‌روزرسانی می‌کند، حتی اگر پیش‌تر رد شده باشد.
-h, --help

راهنمای فرمان deny را نمایش می‌دهد.
-k, --kustomize string

دایرکتوری kustomization را پردازش می‌کند. این فلگ را نمی‌توان همراه با ‎-f یا ‎-R به کار برد.
-o, --output string

قالب خروجی. یکی از این مقادیر: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
-R, --recursive

دایرکتوری مشخص‌شده در ‎-f یا ‎--filename را به‌صورت بازگشتی پردازش می‌کند. این گزینه زمانی سودمند است که بخواهید مانیفست‌های مرتبطی را که درون یک دایرکتوری سازمان‌دهی شده‌اند مدیریت کنید.
--show-managed-fields

اگر مقدار آن true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشته‌ی قالب یا مسیر فایل قالبی که هنگام استفاده از ‎-o=go-template یا ‎-o=go-template-file به کار می‌رود. قالب‌بندی آن مطابق قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که عملیات به جای او (با جعل هویت او) انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات به جای آن انجام می‌شود. برای مشخص کردن چند گروه می‌توانید این فلگ را چند بار تکرار کنید.
--as-uid string

شناسه‌ی UID که عملیات به جای آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات با جعل آن‌ها انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توانید این فلگ را چند بار تکرار کنید.
--cache-dir string     پیش‌فرض: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache)
--certificate-authority string

مسیر فایل گواهی مربوط به مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از فایل kubeconfig که باید استفاده شود
--context string

نام context موجود در فایل kubeconfig که باید استفاده شود
--disable-compression

اگر مقدار آن true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌هایی که به سرور ارسال می‌شوند غیرفعال می‌شود.
--insecure-skip-tls-verify

اگر مقدار آن true باشد، اعتبار گواهی سرور بررسی نمی‌شود. با این کار، اتصال‌های HTTPS شما ناامن می‌شوند.
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های خط فرمان (CLI) استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای تنظیمات ترجیحی استفاده می‌شود. می‌توانید این قابلیت را با export کردن feature gate به صورت KUBECTL_KUBERC=false یا با خاموش کردن قابلیت به صورت KUBERC=off غیرفعال کنید.
--match-server-version

الزام می‌کند که نسخه‌ی سرور با نسخه‌ی کلاینت مطابقت داشته باشد.
-n, --namespace string

در صورت تعیین، دامنه‌ی فضای نام این درخواست خط فرمان را مشخص می‌کند.
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در سرور API
--profile string     پیش‌فرض: "none"

نام پروفایلی که باید ثبت شود. یکی از این مقادیر: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     پیش‌فرض: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی که برای درخواست‌های ارسالی به سرور API استفاده می‌شود
--request-timeout string     پیش‌فرض: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست منفرد به سرور، منتظر می‌ماند. مقادیر غیرصفر باید یکای زمانی متناظر داشته باشند (برای نمونه 1s، 2m، 3h). مقدار صفر به این معناست که درخواست‌ها هیچ‌گاه به پایان مهلت زمانی نمی‌رسند.
-s, --server string

نشانی و پورت سرور API کوبرنتیز
--storage-driver-buffer-duration duration     پیش‌فرض: 1m0s

عملیات نوشتن در درایور ذخیره‌سازی به همین مدت در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در بک‌اندهای غیر از حافظه ثبت می‌شوند.
--storage-driver-db string     پیش‌فرض: "cadvisor"

نام پایگاه داده
--storage-driver-host string     پیش‌فرض: "localhost:8086"

میزبان و پورت پایگاه داده به صورت host:port
--storage-driver-password string     پیش‌فرض: "root"

گذرواژه‌ی پایگاه داده
--storage-driver-secure

--storage-driver-table string     پیش‌فرض: "stats"

نام جدول
--storage-driver-user string     پیش‌فرض: "root"

نام کاربری پایگاه داده

نام سروری که برای اعتبارسنجی گواهی سرور به کار می‌رود. اگر این مقدار ارائه نشود، از نام میزبانی (hostname) استفاده می‌شود که برای برقراری ارتباط با سرور به کار رفته است.
--token string

توکن حامل (bearer token) برای احراز هویت در سرور API
--user string

نام کاربری از فایل kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در سرور API
--version version[=true]

گزینه‌های ‎--version و ‎--version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینه‌ی ‎--version=vX.Y.Z...‎ نسخه‌ی گزارش‌شده را تنظیم می‌کند.
--warnings-as-errors

هشدارهای دریافتی از سرور را به عنوان خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود.
