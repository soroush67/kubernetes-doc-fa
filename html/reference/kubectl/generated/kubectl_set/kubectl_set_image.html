kubectl set image

به‌روزرسانی ایمیج یک الگوی پاد (pod template)

خلاصه

این دستور ایمیج (یا ایمیج‌های) کانتینر موجود در منابع را به‌روزرسانی می‌کند.

منابعی که می‌توان از آن‌ها استفاده کرد عبارت‌اند از (بدون حساسیت به بزرگی و کوچکی حروف):

    pod (po), replicationcontroller (rc), deployment (deploy), daemonset (ds), statefulset (sts), cronjob (cj), replicaset (rs)

kubectl set image (-f FILENAME | TYPE NAME) CONTAINER_NAME_1=CONTAINER_IMAGE_1 ... CONTAINER_NAME_N=CONTAINER_IMAGE_N

مثال‌ها

  # Set a deployment's nginx container image to 'nginx:1.9.1', and its busybox container image to 'busybox'
  kubectl set image deployment/nginx busybox=busybox nginx=nginx:1.9.1
  
  # Update all deployments' and rc's nginx container's image to 'nginx:1.9.1'
  kubectl set image deployments,rc nginx=nginx:1.9.1 --all
  
  # Update image of all containers of daemonset abc to 'nginx:1.9.1'
  kubectl set image daemonset abc *=nginx:1.9.1
  
  # Print result (in yaml format) of updating nginx container image from local file, without hitting the server
  kubectl set image -f path/to/file.yaml nginx=nginx:1.9.1 --local -o yaml

گزینه‌ها

--all

همهٔ منابعِ انواع مشخص‌شده را در فضای نام (namespace) انتخاب می‌کند.
--allow-missing-template-keys     Default: true

اگر مقدار آن true باشد، هرگاه یک فیلد یا کلید نگاشت (map) در الگو وجود نداشته باشد، از خطاهای الگو چشم‌پوشی می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath کاربرد دارد.
--dry-run string[="unchanged"]     Default: "none"

مقدار آن باید یکی از "none"، "server" یا "client" باشد. در راهبرد client، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود و هیچ چیزی ارسال نمی‌شود. در راهبرد server، درخواست در سمت سرور ارسال می‌شود، اما منبع ذخیره و ماندگار نمی‌شود.
--field-manager string     Default: "kubectl-set"

نام مدیری که برای ردگیری مالکیت فیلدها به کار می‌رود.
-f, --filename strings

نام فایل، دایرکتوری یا نشانی URL فایل‌هایی که منبعِ موردنظر برای دریافت از سرور را مشخص می‌کنند.
-h, --help

راهنمای دستور image
-k, --kustomize string

دایرکتوری kustomization را پردازش می‌کند. این پرچم را نمی‌توان همراه با -f یا -R به کار برد.
--local

اگر مقدار آن true باشد، دستور set image با api-server ارتباط برقرار نمی‌کند و به‌صورت محلی اجرا می‌شود.
-o, --output string

قالب خروجی؛ یکی از این موارد: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
-R, --recursive

دایرکتوری‌ای را که در -f, --filename مشخص شده است به‌صورت بازگشتی پردازش می‌کند. این گزینه زمانی سودمند است که بخواهید مانیفست‌های مرتبطی را که در یک دایرکتوری سازمان‌دهی شده‌اند مدیریت کنید.
-l, --selector string

انتخابگر (selector) یا همان پرس‌وجوی برچسب برای پالایش نتایج، که از عملگرهای '='، '=='، '!='، 'in' و 'notin' پشتیبانی می‌کند (برای نمونه: -l key1=value1,key2=value2,key3 in (value3)). آبجکت‌های منطبق باید همهٔ محدودیت‌های برچسبِ مشخص‌شده را برآورده کنند.
--show-managed-fields

اگر مقدار آن true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشتهٔ الگو یا مسیر فایل الگویی که هنگام استفاده از -o=go-template یا -o=go-template-file به کار می‌رود. قالب الگو همان الگوهای golang است [http://golang.org/pkg/text/template/#pkg-overview].

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری‌ای که عملیات با جعل هویت (impersonate) آن انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام باشد.
--as-group strings

گروهی که عملیات با جعل هویت آن انجام می‌شود. برای مشخص کردن چند گروه می‌توان این پرچم را چند بار تکرار کرد.
--as-uid string

شناسهٔ UID که عملیات با جعل هویت آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات با جعل هویت آن انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توان این پرچم را چند بار تکرار کرد.
--cache-dir string     Default: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظهٔ نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام زمینه (context) ای از kubeconfig که باید استفاده شود
--disable-compression

اگر مقدار آن true باشد، فشرده‌سازی پاسخ برای همهٔ درخواست‌های ارسالی به سرور غیرفعال می‌شود.
--insecure-skip-tls-verify

اگر مقدار آن true باشد، اعتبار گواهی سرور بررسی نمی‌شود. این کار اتصال‌های HTTPS شما را ناامن می‌کند.
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های خط فرمان (CLI) استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات کاربر استفاده می‌شود. می‌توان این قابلیت را با export کردن feature gate به‌صورت KUBECTL_KUBERC=false یا با خاموش کردن قابلیت به‌صورت KUBERC=off غیرفعال کرد.
--match-server-version

الزام می‌کند که نسخهٔ سرور با نسخهٔ کلاینت مطابقت داشته باشد.
-n, --namespace string

در صورت تعیین، دامنهٔ فضای نام برای این درخواست خط فرمان را مشخص می‌کند.
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در سرور API
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود؛ یکی از این موارد: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی که برای درخواست‌های ارسالی به سرور API استفاده می‌شود
--request-timeout string     Default: "0"

مدت زمانی که پیش از رها کردن یک درخواست منفرد به سرور باید منتظر ماند. مقادیر غیرصفر باید واحد زمانی متناظر را داشته باشند (برای نمونه 1s، 2m یا 3h). مقدار صفر به این معناست که برای درخواست‌ها مهلت زمانی در نظر گرفته نمی‌شود.
-s, --server string

نشانی و درگاه (port) سرور API کوبرنتیز
--storage-driver-buffer-duration duration     Default: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به اندازهٔ این مدت در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای (non memory backends) ثبت می‌شوند.
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده
--storage-driver-host string     Default: "localhost:8086"

میزبان و درگاه پایگاه داده به‌صورت host:port
--storage-driver-password string     Default: "root"

گذرواژهٔ پایگاه داده
--storage-driver-secure

از اتصال امن با پایگاه داده استفاده می‌کند
--storage-driver-table string     Default: "stats"

نام جدول
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر این مقدار ارائه نشود، از نام میزبانی (hostname) استفاده می‌شود که برای برقراری ارتباط با سرور به کار رفته است.
--token string

توکن حامل (bearer token) برای احراز هویت در سرور API
--user string

نام کاربری از kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در سرور API
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینهٔ --version=vX.Y.Z... نسخهٔ گزارش‌شده را تنظیم می‌کند.
--warnings-as-errors

هشدارهای دریافتی از سرور را خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود.
