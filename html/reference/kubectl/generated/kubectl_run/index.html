kubectl run

اجرای یک ایمیج مشخص در کلاستر

خلاصه

این فرمان یک ایمیج مشخص را در یک پاد ایجاد و اجرا می‌کند.

kubectl run NAME --image=image [--env="key=value"] [--port=port] [--dry-run=server|client] [--overrides=inline-json] [--command] -- [COMMAND] [args...]

مثال‌ها

  # Start a nginx pod
  kubectl run nginx --image=nginx
  
  # Start a hazelcast pod and let the container expose port 5701
  kubectl run hazelcast --image=hazelcast/hazelcast --port=5701
  
  # Start a hazelcast pod and set environment variables "DNS_DOMAIN=cluster" and "POD_NAMESPACE=default" in the container
  kubectl run hazelcast --image=hazelcast/hazelcast --env="DNS_DOMAIN=cluster" --env="POD_NAMESPACE=default"
  
  # Start a hazelcast pod and set labels "app=hazelcast" and "env=prod" in the container
  kubectl run hazelcast --image=hazelcast/hazelcast --labels="app=hazelcast,env=prod"
  
  # Dry run; print the corresponding API objects without creating them
  kubectl run nginx --image=nginx --dry-run=client
  
  # Start a nginx pod, but overload the spec with a partial set of values parsed from JSON
  kubectl run nginx --image=nginx --overrides='{ "apiVersion": "v1", "spec": { ... } }'
  
  # Start a busybox pod and keep it in the foreground, don't restart it if it exits
  kubectl run -i -t busybox --image=busybox --restart=Never
  
  # Start the nginx pod using the default command, but use custom arguments (arg1 .. argN) for that command
  kubectl run nginx --image=nginx -- <arg1> <arg2> ... <argN>
  
  # Start the nginx pod using a different command and custom arguments
  kubectl run nginx --image=nginx --command -- <cmd> <arg1> ... <argN>

گزینه‌ها

--allow-missing-template-keys     Default: true

اگر true باشد، هنگامی که یک فیلد یا کلید map در قالب وجود نداشته باشد، از هرگونه خطا در قالب‌ها چشم‌پوشی می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath کاربرد دارد.
--annotations strings

حاشیه‌نویسی‌هایی (annotation) که روی پاد اعمال می‌شوند.
--attach

اگر true باشد، منتظر می‌ماند تا اجرای پاد آغاز شود و سپس، مانند زمانی که 'kubectl attach ...' فراخوانی شده باشد، به پاد متصل می‌شود. مقدار پیش‌فرض false است، مگر اینکه '-i/--stdin' تنظیم شده باشد که در این صورت مقدار پیش‌فرض true خواهد بود. با '--restart=Never'، کد خروج فرایند کانتینر بازگردانده می‌شود.
--cascade string[="background"]     Default: "background"

مقدار باید یکی از "background"، "orphan" یا "foreground" باشد. این گزینه راهبرد حذف آبشاری (cascading) را برای وابسته‌ها (برای مثال، پادهایی که یک ReplicationController ایجاد کرده است) تعیین می‌کند. مقدار پیش‌فرض background است.
--command

اگر true باشد و آرگومان‌های اضافی وجود داشته باشند، از آن‌ها به‌عنوان فیلد 'command' در کانتینر استفاده می‌شود، نه فیلد 'args' که حالت پیش‌فرض است.
--detach-keys string     Default: "ctrl-p,ctrl-q"

توالی کلیدهای جداشدن از کانتینر را بازنویسی می‌کند.
--dry-run string[="unchanged"]     Default: "none"

مقدار باید یکی از "none"، "server" یا "client" باشد. در راهبرد client، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود، بی‌آنکه واقعاً ارسال شود. در راهبرد server، درخواست به سمت سرور فرستاده می‌شود، بدون اینکه منبع به‌طور ماندگار ذخیره شود.
--env strings

متغیرهای محیطی که در کانتینر تنظیم می‌شوند.
--expose --port

اگر true باشد، یک Service از نوع ClusterIP مرتبط با پاد ایجاد می‌شود. این گزینه به --port نیاز دارد.
--field-manager string     Default: "kubectl-run"

نام مدیری (manager) که برای ردیابی مالکیت فیلدها استفاده می‌شود.
--force

اگر true باشد، منابع بی‌درنگ از API حذف می‌شوند و حذف تدریجی (graceful deletion) نادیده گرفته می‌شود. توجه کنید که حذف فوری برخی منابع ممکن است به ناسازگاری یا از دست رفتن داده‌ها منجر شود و به تأیید نیاز دارد.
--grace-period int     Default: -1

مدت زمانی بر حسب ثانیه که به منبع داده می‌شود تا به‌صورت تدریجی خاتمه یابد. اگر مقدار منفی باشد، نادیده گرفته می‌شود. برای خاموش‌شدن فوری، آن را روی 1 تنظیم کنید. فقط زمانی می‌توان آن را روی 0 تنظیم کرد که --force برابر true باشد (حذف اجباری).
-h, --help

راهنمای run
--image string

ایمیجی که کانتینر باید اجرا کند.
--image-pull-policy string

سیاست دریافت (pull) ایمیج برای کانتینر. اگر خالی بماند، کلاینت این مقدار را مشخص نمی‌کند و سرور مقدار پیش‌فرض را برای آن تعیین می‌کند.
-k, --kustomize string

یک دایرکتوری kustomization را پردازش می‌کند. این فلگ را نمی‌توان همراه با -f یا -R استفاده کرد.
-l, --labels string

برچسب‌هایی که با کاما از هم جدا شده‌اند و روی پاد اعمال می‌شوند. این برچسب‌ها مقادیر قبلی را بازنویسی می‌کنند.
--leave-stdin-open

اگر پاد در حالت تعاملی یا با stdin آغاز شده باشد، پس از پایان نخستین اتصال (attach)، stdin باز می‌ماند. به‌طور پیش‌فرض، stdin پس از پایان نخستین اتصال بسته می‌شود.
-o, --output string

قالب خروجی. یکی از این موارد: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
--override-type string     Default: "merge"

روشی که برای بازنویسی آبجکت تولیدشده به کار می‌رود: json، merge یا strategic.
--overrides string

یک بازنویسی JSON درون‌خطی برای آبجکت تولیدشده. اگر این مقدار خالی نباشد، از آن برای بازنویسی آبجکت تولیدشده استفاده می‌شود. لازم است آبجکت یک فیلد apiVersion معتبر داشته باشد.
--pod-running-timeout duration     Default: 1m0s

مدت زمانی (مانند 5s، 2m یا 3h و بزرگ‌تر از صفر) که باید منتظر ماند تا دست‌کم یک پاد در حال اجرا باشد
--port string

پورتی که این کانتینر در دسترس قرار می‌دهد.
--privileged

اگر true باشد، کانتینر در حالت ممتاز (privileged) اجرا می‌شود.
-q, --quiet

اگر true باشد، پیام‌های اعلان (prompt) نمایش داده نمی‌شوند.
-R, --recursive

دایرکتوری مشخص‌شده در -f, --filename را به‌صورت بازگشتی پردازش می‌کند. این گزینه زمانی مفید است که بخواهید مانیفست‌های مرتبطی را که در یک دایرکتوری سازمان‌دهی شده‌اند مدیریت کنید.
--restart string     Default: "Always"

سیاست راه‌اندازی مجدد این پاد. مقادیر مجاز: [Always, OnFailure, Never].
--rm

اگر true باشد، پاد پس از خروج حذف می‌شود. این گزینه فقط هنگام اتصال به کانتینر معتبر است؛ برای مثال با '--attach' یا '-i/--stdin'.
--save-config

اگر true باشد، پیکربندی آبجکت فعلی در حاشیه‌نویسی آن ذخیره می‌شود. در غیر این صورت، حاشیه‌نویسی بدون تغییر باقی می‌ماند. این فلگ زمانی مفید است که بخواهید در آینده kubectl apply را روی این آبجکت اجرا کنید.
--show-managed-fields

اگر true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
-i, --stdin

ورودی استاندارد (stdin) کانتینر درون پاد را باز نگه می‌دارد، حتی اگر چیزی به آن متصل نباشد.
--template string

رشته قالب یا مسیر فایل قالب که هنگام استفاده از -o=go-template یا -o=go-template-file به کار می‌رود. قالب‌ها از نوع قالب‌های golang هستند [http://golang.org/pkg/text/template/#pkg-overview].
--timeout duration

مدت زمان انتظار پیش از صرف‌نظر کردن از یک عملیات حذف؛ مقدار صفر یعنی مهلت زمانی بر اساس اندازه آبجکت تعیین شود
-t, --tty

یک TTY برای کانتینر درون پاد تخصیص می‌دهد.
--wait

اگر true باشد، پیش از بازگشت منتظر می‌ماند تا منابع حذف شوند. این گزینه منتظر اجرای finalizerها نیز می‌ماند.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری‌ای که عملیات با جعل هویت (impersonate) آن انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات با جعل هویت آن انجام می‌شود. برای مشخص کردن چند گروه می‌توانید این فلگ را چند بار تکرار کنید.
--as-uid string

شناسهٔ UID که عملیات با جعل هویت آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات با جعل هویت آن انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توانید این فلگ را چند بار تکرار کنید.
--cache-dir string     پیش‌فرض: "$HOME/.kube/cache"

پوشهٔ پیش‌فرض حافظهٔ نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام زمینه (context) از kubeconfig که باید استفاده شود
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همهٔ درخواست‌ها به سرور غیرفعال می‌شود
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نمی‌شود. با این کار اتصال‌های HTTPS شما ناامن می‌شوند
--kubeconfig string

مسیر فایل kubeconfig برای استفاده در درخواست‌های CLI.
--kuberc string

مسیر فایل kuberc برای استفاده در ترجیحات. می‌توان این قابلیت را با export کردن feature gate به‌صورت KUBECTL_KUBERC=false یا خاموش کردن قابلیت با KUBERC=off غیرفعال کرد.
--match-server-version

الزام به یکسان بودن نسخهٔ سرور با نسخهٔ کلاینت
-n, --namespace string

در صورت تعیین، دامنهٔ فضای نام برای این درخواست CLI
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در API server
--profile string     پیش‌فرض: "none"

نام پروفایلی که باید ثبت شود. یکی از مقادیر (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     پیش‌فرض: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی پراکسی برای استفاده در درخواست‌ها به API server
--request-timeout string     پیش‌فرض: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست به سرور باید منتظر ماند. مقادیر غیرصفر باید واحد زمانی متناظر داشته باشند (مثلاً 1s، 2m، 3h). مقدار صفر یعنی درخواست‌ها هیچ‌گاه به مهلت زمانی نمی‌رسند.
-s, --server string

نشانی و درگاه API server کوبرنتیز
--storage-driver-buffer-duration duration     پیش‌فرض: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به همین مدت در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای ثبت می‌شوند
--storage-driver-db string     پیش‌فرض: "cadvisor"

نام پایگاه داده
--storage-driver-host string     پیش‌فرض: "localhost:8086"

میزبان و درگاه پایگاه داده به‌صورت host:port
--storage-driver-password string     پیش‌فرض: "root"

گذرواژهٔ پایگاه داده
--storage-driver-secure

استفاده از اتصال امن با پایگاه داده
--storage-driver-table string     پیش‌فرض: "stats"

نام جدول
--storage-driver-user string     پیش‌فرض: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر تعیین نشود، نام میزبانی که برای ارتباط با سرور به کار رفته است استفاده می‌شود
--token string

توکن حامل (bearer token) برای احراز هویت در API server
--user string

نام کاربری از kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در API server
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینهٔ --version=vX.Y.Z... نسخهٔ گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافتی از سرور به‌عنوان خطا در نظر گرفته می‌شوند و برنامه با کد خروج غیرصفر خارج می‌شود
