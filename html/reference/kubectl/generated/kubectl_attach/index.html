kubectl attach

اتصال به یک کانتینر در حال اجرا

خلاصه

این دستور به فرایندی متصل می‌شود که از پیش درون یک کانتینر موجود در حال اجرا است.

kubectl attach (POD | TYPE/NAME) -c CONTAINER

مثال‌ها

  # Get output from running pod mypod; use the 'kubectl.kubernetes.io/default-container' annotation
  # for selecting the container to be attached or the first container in the pod will be chosen
  kubectl attach mypod
  
  # Get output from ruby-container from pod mypod
  kubectl attach mypod -c ruby-container
  
  # Switch to raw terminal mode; sends stdin to 'bash' in ruby-container from pod mypod
  # and sends stdout/stderr from 'bash' back to the client
  kubectl attach mypod -c ruby-container -i -t
  
  # Get output from the first pod of a replica set named nginx
  kubectl attach rs/nginx

گزینه‌ها

-c, --container string

نام کانتینر. اگر این مقدار مشخص نشود، برای انتخاب کانتینری که باید به آن متصل شد از حاشیه‌نویسی (annotation) kubectl.kubernetes.io/default-container استفاده می‌شود؛ در غیر این صورت، نخستین کانتینر موجود در پاد انتخاب خواهد شد.
--detach-keys string     Default: "ctrl-p,ctrl-q"

دنباله‌ی کلیدهایی را که برای جدا شدن از یک کانتینر به کار می‌رود، با مقدار دلخواه جایگزین می‌کند.
-h, --help

راهنمای دستور attach را نمایش می‌دهد.
--pod-running-timeout duration     Default: 1m0s

مدت زمانی (مانند 5s، 2m یا 3h و بزرگ‌تر از صفر) که باید منتظر ماند تا دست‌کم یک پاد در حال اجرا باشد.
-q, --quiet

فقط خروجی نشست راه دور را چاپ می‌کند.
-i, --stdin

ورودی استاندارد (stdin) را به کانتینر ارسال می‌کند.
-t, --tty

مشخص می‌کند که ورودی استاندارد یک ترمینال (TTY) است.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که عملیات از طرف او و با هویت او انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات با هویت آن انجام می‌شود. برای مشخص کردن چند گروه می‌توانید این فلگ را چند بار تکرار کنید.
--as-uid string

شناسه‌ی یکتا (UID) که عملیات با هویت آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که عملیات با هویت آن انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توانید این فلگ را چند بار تکرار کنید.
--cache-dir string     Default: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache).
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority).
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS.
--client-key string

مسیر فایل کلید کلاینت برای TLS.
--cluster string

نام کلاستری از فایل kubeconfig که باید استفاده شود.
--context string

نام زمینه (context) ای از فایل kubeconfig که باید استفاده شود.
--disable-compression

اگر مقدار آن true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌هایی که به سرور فرستاده می‌شوند غیرفعال می‌شود.
--insecure-skip-tls-verify

اگر مقدار آن true باشد، اعتبار گواهی سرور بررسی نمی‌شود. در این حالت، اتصال‌های HTTPS شما ناامن خواهند بود.
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های خط فرمان (CLI) استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات کاربر استفاده می‌شود. برای غیرفعال کردن آن می‌توانید feature gate مربوط را با اکسپورت کردن KUBECTL_KUBERC=false غیرفعال کنید یا قابلیت را با KUBERC=off خاموش کنید.
--match-server-version

الزام می‌کند که نسخه‌ی سرور با نسخه‌ی کلاینت مطابقت داشته باشد.
-n, --namespace string

در صورت تعیین، دامنه‌ی فضای نام این درخواست خط فرمان را مشخص می‌کند.
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در سرور API.
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود. یکی از این مقادیر است: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود.
--proxy-url string

نشانی URL پراکسی که برای درخواست‌ها به سرور API استفاده می‌شود.
--request-timeout string     Default: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست منفرد به سرور باید منتظر ماند. مقادیر غیرصفر باید واحد زمانی متناظر را داشته باشند (مثلاً 1s، 2m یا 3h). مقدار صفر به این معناست که برای درخواست‌ها هیچ مهلت زمانی‌ای در نظر گرفته نمی‌شود.
-s, --server string

نشانی و پورت سرور API کوبرنتیز.
--storage-driver-buffer-duration duration     Default: 1m0s

عملیات نوشتن در درایور ذخیره‌سازی به مدت این بازه‌ی زمانی در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای (non memory backends) ثبت می‌شوند.
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده.
--storage-driver-host string     Default: "localhost:8086"

میزبان و پورت پایگاه داده به شکل host:port.
--storage-driver-password string     Default: "root"

گذرواژه‌ی پایگاه داده.
--storage-driver-secure

برای ارتباط با پایگاه داده از اتصال امن استفاده می‌کند.
--storage-driver-table string     Default: "stats"

نام جدول.
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده.
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر این مقدار ارائه نشود، از نام میزبانی که برای برقراری ارتباط با سرور به کار رفته است استفاده می‌شود.
--token string

توکن حامل (bearer token) برای احراز هویت در سرور API.
--user string

نام کاربری از فایل kubeconfig که باید استفاده شود.
--username string

نام کاربری برای احراز هویت پایه در سرور API.
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینه‌ی --version=vX.Y.Z... نسخه‌ی گزارش‌شده را تعیین می‌کند.
--warnings-as-errors

هشدارهای دریافت‌شده از سرور را به‌عنوان خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود.
