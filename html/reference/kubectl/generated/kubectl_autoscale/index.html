kubectl autoscale

مقیاس‌دهی خودکار یک Deployment، ReplicaSet، StatefulSet یا ReplicationController

خلاصه

این دستور یک مقیاس‌دهنده‌ی خودکار (autoscaler) ایجاد می‌کند که تعداد پادهای در حال اجرا در یک کلاستر کوبرنتیز را به‌طور خودکار انتخاب و تنظیم می‌کند. دستور ابتدا تلاش می‌کند از API نسخه‌ی autoscaling/v2 استفاده کند و در صورت بروز خطا، به API نسخه‌ی autoscaling/v1 بازمی‌گردد.

این دستور یک Deployment، ReplicaSet، StatefulSet یا ReplicationController را بر اساس نام جست‌وجو می‌کند و یک مقیاس‌دهنده‌ی خودکار می‌سازد که از منبع داده‌شده به‌عنوان مرجع استفاده می‌کند. مقیاس‌دهنده‌ی خودکار می‌تواند تعداد پادهای مستقر در سیستم را در صورت نیاز به‌طور خودکار افزایش یا کاهش دهد.

kubectl autoscale (-f FILENAME | TYPE NAME | TYPE/NAME) [--min=MINPODS] --max=MAXPODS [--cpu=CPU] [--memory=MEMORY]

مثال‌ها

  # Auto scale a deployment "foo", with the number of pods between 2 and 10, no target CPU utilization specified so a default autoscaling policy will be used
  kubectl autoscale deployment foo --min=2 --max=10
  
  # Auto scale a replication controller "foo", with the number of pods between 1 and 5, target CPU utilization at 80%
  kubectl autoscale rc foo --max=5 --cpu=80%
  
  # Auto scale a deployment "bar", with the number of pods between 3 and 6, target average CPU of 500m and memory of 200Mi
  kubectl autoscale deployment bar --min=3 --max=6 --cpu=500m --memory=200Mi
  
  # Auto scale a deployment "bar", with the number of pods between 2 and 8, target CPU utilization 60% and memory utilization 70%
  kubectl autoscale deployment bar --min=2 --max=8 --cpu=60% --memory=70%

گزینه‌ها

--allow-missing-template-keys     Default: true

اگر true باشد، هر خطایی که به‌دلیل نبودِ یک فیلد یا کلید نگاشت در قالب رخ دهد نادیده گرفته می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath اعمال می‌شود.
--cpu string

میزان هدف مصرف CPU در کل پادها. اگر به‌صورت درصد مشخص شود (برای مثال "70%" برای ۷۰ درصد از CPU درخواست‌شده)، میانگین مصرف هدف قرار می‌گیرد. اگر به‌صورت مقدار مشخص شود (برای مثال "500m" برای ۵۰۰ میلی‌CPU)، میانگین مقدار هدف قرار می‌گیرد. مقدار بدون واحد به‌عنوان مقداری با واحد میلی‌CPU در نظر گرفته می‌شود (برای مثال "500" همان "500m" است).
--dry-run string[="unchanged"]     Default: "none"

مقدار آن باید یکی از "none"، "server" یا "client" باشد. در راهبرد client، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود و ارسالی صورت نمی‌گیرد. در راهبرد server، درخواست سمت سرور ارسال می‌شود، اما منبع ذخیره نمی‌شود.
--field-manager string     Default: "kubectl-autoscale"

نام مدیری که برای ردیابی مالکیت فیلدها استفاده می‌شود.
-f, --filename strings

نام فایل، دایرکتوری یا URL فایل‌هایی که منبعِ موردنظر برای مقیاس‌دهی خودکار را مشخص می‌کنند.
-h, --help

راهنمای autoscale
-k, --kustomize string

دایرکتوری kustomization را پردازش می‌کند. این فلگ را نمی‌توان همراه با -f یا -R به کار برد.
--max int32     Default: -1

حد بالای تعداد پادهایی که مقیاس‌دهنده‌ی خودکار می‌تواند تنظیم کند. این فلگ الزامی است.
--memory string

میزان هدف مصرف حافظه در کل پادها. اگر به‌صورت درصد مشخص شود (برای مثال "60%" برای ۶۰ درصد از حافظه‌ی درخواست‌شده)، میانگین مصرف هدف قرار می‌گیرد. اگر به‌صورت مقدار مشخص شود (برای مثال "200Mi" برای ۲۰۰ مبی‌بایت و "1Gi" برای ۱ گیبی‌بایت)، میانگین مقدار هدف قرار می‌گیرد. مقدار بدون واحد به‌عنوان مقداری با واحد مبی‌بایت در نظر گرفته می‌شود (برای مثال "200" همان "200Mi" است).
--min int32     Default: -1

حد پایین تعداد پادهایی که مقیاس‌دهنده‌ی خودکار می‌تواند تنظیم کند. اگر مشخص نشود یا منفی باشد، سرور یک مقدار پیش‌فرض اعمال می‌کند.
--name string

نام آبجکتِ تازه ایجادشده. اگر مشخص نشود، از نام منبع ورودی استفاده می‌شود.
-o, --output string

قالب خروجی. یکی از این مقادیر: (json, yaml, kyaml, name, go-template, go-template-file, template, templatefile, jsonpath, jsonpath-as-json, jsonpath-file).
-R, --recursive

دایرکتوری مشخص‌شده در -f یا --filename را به‌صورت بازگشتی پردازش می‌کند. این گزینه زمانی مفید است که بخواهید مانیفست‌های مرتبطی را که در یک دایرکتوری سازمان‌دهی شده‌اند مدیریت کنید.
--save-config

اگر true باشد، پیکربندی آبجکت فعلی در حاشیه‌نویسی (annotation) آن ذخیره می‌شود؛ در غیر این صورت، حاشیه‌نویسی بدون تغییر باقی می‌ماند. این فلگ زمانی مفید است که بخواهید در آینده دستور kubectl apply را روی این آبجکت اجرا کنید.
--show-managed-fields

اگر true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشته‌ی قالب یا مسیر فایل قالب برای استفاده در حالت‌های -o=go-template و -o=go-template-file. قالب‌ها از نوع قالب‌های golang هستند [http://golang.org/pkg/text/template/#pkg-overview].

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری‌ای که عملیات با هویت آن انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات با هویت آن انجام می‌شود. برای مشخص‌کردن چند گروه می‌توان این فلگ را تکرار کرد.
--as-uid string

شناسه‌ی UID که عملیات با هویت آن انجام می‌شود.
--as-user-extra strings

اطلاعات اضافی کاربر که عملیات با هویت آن انجام می‌شود. برای مشخص‌کردن چند مقدار برای یک کلید می‌توان این فلگ را تکرار کرد.
--cache-dir string     Default: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض کش
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام زمینه‌ای (context) از kubeconfig که باید استفاده شود
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌ها به سرور غیرفعال می‌شود
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نمی‌شود. این کار اتصال‌های HTTPS شما را ناامن می‌کند
--kubeconfig string

مسیر فایل kubeconfig برای استفاده در درخواست‌های CLI.
--kuberc string

مسیر فایل kuberc برای استفاده در تنظیمات ترجیحی. می‌توان این قابلیت را با export کردن feature gate به‌صورت KUBECTL_KUBERC=false یا خاموش‌کردن قابلیت با KUBERC=off غیرفعال کرد.
--match-server-version

الزام به یکسان‌بودن نسخه‌ی سرور با نسخه‌ی کلاینت
-n, --namespace string

در صورت وجود، دامنه‌ی فضای نام برای این درخواست CLI
--password string

گذرواژه برای احراز هویت پایه در سرور API
--profile string     Default: "none"

نام پروفایلی که باید ثبت شود. یکی از این مقادیر: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     Default: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی برای استفاده در درخواست‌ها به سرور API
--request-timeout string     Default: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست به سرور منتظر می‌ماند. مقادیر غیرصفر باید واحد زمانی متناظر داشته باشند (برای مثال 1s، 2m، 3h). مقدار صفر به این معناست که درخواست‌ها هیچ مهلت زمانی ندارند.
-s, --server string

نشانی و پورت سرور API کوبرنتیز
--storage-driver-buffer-duration duration     Default: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به این مدت بافر می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای ثبت می‌شوند
--storage-driver-db string     Default: "cadvisor"

نام پایگاه داده
--storage-driver-host string     Default: "localhost:8086"

نشانی host:port پایگاه داده
--storage-driver-password string     Default: "root"

گذرواژه‌ی پایگاه داده
--storage-driver-secure

استفاده از اتصال امن با پایگاه داده
--storage-driver-table string     Default: "stats"

نام جدول
--storage-driver-user string     Default: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر ارائه نشود، از نام میزبانی که برای برقراری ارتباط با سرور به کار رفته است استفاده می‌شود
--token string

توکن حامل (bearer token) برای احراز هویت در سرور API
--user string

نام کاربری از kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در سرور API
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و خارج می‌شوند؛ گزینه‌ی --version=vX.Y.Z... نسخه‌ی گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافتی از سرور را خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود
