kubectl create secret generic

ایجاد یک Secret از یک فایل محلی، یک دایرکتوری یا یک مقدار لفظی

خلاصه

یک Secret را بر اساس یک فایل، یک دایرکتوری یا یک مقدار لفظی (literal) مشخص ایجاد می‌کند.

هر Secret می‌تواند یک یا چند جفت کلید/مقدار را در خود بسته‌بندی کند.

هنگام ایجاد Secret بر اساس یک فایل، کلید به‌طور پیش‌فرض نام پایه (basename) فایل و مقدار به‌طور پیش‌فرض محتوای فایل خواهد بود. اگر نام پایه کلید معتبری نباشد یا بخواهید کلید دلخواه خود را انتخاب کنید، می‌توانید کلید دیگری مشخص کنید.

هنگام ایجاد Secret بر اساس یک دایرکتوری، هر فایلی از آن دایرکتوری که نام پایه‌اش کلید معتبری باشد، در Secret بسته‌بندی می‌شود. همه‌ی ورودی‌های دایرکتوری به‌جز فایل‌های معمولی نادیده گرفته می‌شوند (مانند زیردایرکتوری‌ها، پیوندهای نمادین، دستگاه‌ها، لوله‌ها و غیره).

kubectl create secret generic NAME [--type=string] [--from-file=[key=]source] [--from-literal=key1=value1] [--dry-run=server|client|none]

مثال‌ها

  # Create a new secret named my-secret with keys for each file in folder bar
  kubectl create secret generic my-secret --from-file=path/to/bar
  
  # Create a new secret named my-secret with specified keys instead of names on disk
  kubectl create secret generic my-secret --from-file=ssh-privatekey=path/to/id_rsa --from-file=ssh-publickey=path/to/id_rsa.pub
  
  # Create a new secret named my-secret with key1=supersecret and key2=topsecret
  kubectl create secret generic my-secret --from-literal=key1=supersecret --from-literal=key2=topsecret
  
  # Create a new secret named my-secret using a combination of a file and a literal
  kubectl create secret generic my-secret --from-file=ssh-privatekey=path/to/id_rsa --from-literal=passphrase=topsecret
  
  # Create a new secret named my-secret from env files
  kubectl create secret generic my-secret --from-env-file=path/to/foo.env --from-env-file=path/to/bar.env

گزینه‌ها

--allow-missing-template-keys     پیش‌فرض: true

اگر true باشد، هرگاه فیلد یا کلید نگاشتی در قالب وجود نداشته باشد، خطاهای قالب نادیده گرفته می‌شوند. این گزینه فقط برای قالب‌های خروجی golang و jsonpath اعمال می‌شود.
--append-hash

یک هش از Secret را به انتهای نام آن اضافه می‌کند.
--dry-run string[="unchanged"]     پیش‌فرض: "none"

مقدار آن باید یکی از "none"، "server" یا "client" باشد. در راهبرد client، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود، بدون آنکه ارسال شود. در راهبرد server، درخواست سمت سرور ارسال می‌شود، بدون آنکه منبع ذخیره شود.
--field-manager string     پیش‌فرض: "kubectl-create"

نام مدیری که برای ردیابی مالکیت فیلدها استفاده می‌شود.
--from-env-file strings

مسیر فایلی را مشخص می‌کند که خطوط جفت‌های key=val از آن خوانده می‌شوند تا Secret ایجاد شود.
--from-file strings

فایل‌های کلید را می‌توان با مسیر فایلشان مشخص کرد که در این صورت نامی پیش‌فرض به آن‌ها داده می‌شود، یا به‌صورت اختیاری با یک نام و مسیر فایل مشخص کرد که در این صورت از نام داده‌شده استفاده می‌شود. اگر یک دایرکتوری مشخص شود، روی همه‌ی فایل‌های نام‌دار آن دایرکتوری که کلید معتبری برای Secret باشند پیمایش انجام می‌شود.
--from-literal strings

یک کلید و مقدار لفظی را برای درج در Secret مشخص می‌کند (مثلاً mykey=somevalue)
-h, --help

راهنمای فرمان generic
-o, --output string

قالب خروجی؛ یکی از این مقادیر: (json، yaml، kyaml، name، go-template، go-template-file، template، templatefile، jsonpath، jsonpath-as-json، jsonpath-file).
--save-config

اگر true باشد، پیکربندی آبجکت فعلی در حاشیه‌نویسی (annotation) آن ذخیره می‌شود؛ در غیر این صورت، حاشیه‌نویسی بدون تغییر می‌ماند. این فلگ زمانی مفید است که بخواهید در آینده فرمان kubectl apply را روی این آبجکت اجرا کنید.
--show-managed-fields

اگر true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشته‌ی قالب یا مسیر فایل قالب که هنگام استفاده از ‎-o=go-template یا ‎-o=go-template-file به کار می‌رود. قالب‌بندی آن مطابق قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].
--type string

نوع Secret مورد نظر برای ایجاد
--validate string[="strict"]     پیش‌فرض: "strict"

مقدار آن باید یکی از این موارد باشد: strict (یا true)، warn، ignore (یا false). با مقدار "true" یا "strict"، ورودی با یک طرح‌واره (schema) اعتبارسنجی می‌شود و در صورت نامعتبر بودن، درخواست با شکست مواجه می‌شود. اگر ServerSideFieldValidation روی api-server فعال باشد، اعتبارسنجی سمت سرور انجام می‌شود و در غیر این صورت، به اعتبارسنجی کم‌اطمینان‌تر سمت کلاینت بازمی‌گردد. با مقدار "warn"، اگر اعتبارسنجی فیلد سمت سرور روی API server فعال باشد، درباره‌ی فیلدهای ناشناخته یا تکراری هشدار داده می‌شود بدون آنکه درخواست مسدود شود، و در غیر این صورت رفتاری مانند "ignore" خواهد داشت. با مقدار "false" یا "ignore"، هیچ اعتبارسنجی طرح‌واره‌ای انجام نمی‌شود و فیلدهای ناشناخته یا تکراری بی‌صدا حذف می‌شوند.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که برای این عملیات جعل هویت (impersonate) می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که برای این عملیات جعل هویت می‌شود. برای مشخص کردن چند گروه، می‌توان این فلگ را تکرار کرد.
--as-uid string

شناسه‌ی UID که برای این عملیات جعل هویت می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربر که برای این عملیات جعل هویت می‌شود. برای مشخص کردن چند مقدار برای یک کلید، می‌توان این فلگ را تکرار کرد.
--cache-dir string     پیش‌فرض: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستر kubeconfig مورد استفاده
--context string

نام context مورد استفاده در kubeconfig
--disable-compression

اگر true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌ها به سرور غیرفعال می‌شود.
--insecure-skip-tls-verify

اگر true باشد، اعتبار گواهی سرور بررسی نمی‌شود. این کار اتصال‌های HTTPS شما را ناامن می‌کند.
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های CLI استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات استفاده می‌شود. می‌توان این قابلیت را با اکسپورت کردن feature gate ‏KUBECTL_KUBERC=false یا خاموش کردن قابلیت KUBERC=off غیرفعال کرد.
--match-server-version

الزام می‌کند که نسخه‌ی سرور با نسخه‌ی کلاینت مطابقت داشته باشد.
-n, --namespace string

در صورت تعیین، دامنه‌ی فضای نام برای این درخواست CLI
--password string

گذرواژه برای احراز هویت پایه در API server
--profile string     پیش‌فرض: "none"

نام پروفایلی که باید ثبت شود؛ یکی از (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     پیش‌فرض: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی پراکسی برای درخواست‌ها به API server
--request-timeout string     پیش‌فرض: "0"

مدت زمان انتظار پیش از صرف‌نظر کردن از یک درخواست منفرد به سرور. مقادیر غیرصفر باید واحد زمانی متناظر داشته باشند (مثلاً 1s، 2m، 3h). مقدار صفر به این معناست که برای درخواست‌ها مهلت زمانی در نظر گرفته نمی‌شود.
-s, --server string

نشانی و درگاه API server کوبرنتیز
--storage-driver-buffer-duration duration     پیش‌فرض: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به این مدت بافر می‌شوند و سپس در قالب یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای ثبت می‌شوند.
--storage-driver-db string     پیش‌فرض: "cadvisor"

نام پایگاه داده
--storage-driver-host string     پیش‌فرض: "localhost:8086"

میزبان و درگاه پایگاه داده (host:port)
--storage-driver-password string     پیش‌فرض: "root"

گذرواژه‌ی پایگاه داده
--storage-driver-secure

استفاده از اتصال امن با پایگاه داده
--storage-driver-table string     پیش‌فرض: "stats"

نام جدول
--storage-driver-user string     پیش‌فرض: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر تعیین نشود، از نام میزبانی که برای برقراری ارتباط با سرور به کار رفته استفاده می‌شود.
--token string

توکن حامل (bearer token) برای احراز هویت در API server
--user string

نام کاربر kubeconfig مورد استفاده
--username string

نام کاربری برای احراز هویت پایه در API server
--version version[=true]

فلگ‌های ‎--version و ‎--version=raw اطلاعات نسخه را چاپ می‌کنند و خارج می‌شوند؛ فلگ ‎--version=vX.Y.Z...‎ نسخه‌ی گزارش‌شده را تنظیم می‌کند.
--warnings-as-errors

هشدارهای دریافتی از سرور را خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود.
