kubectl create namespace

ایجاد یک فضای نام (namespace) با نام مشخص‌شده

خلاصه

این دستور یک فضای نام (namespace) با نام مشخص‌شده ایجاد می‌کند.

kubectl create namespace NAME [--dry-run=server|client|none]

مثال‌ها

  # Create a new namespace named my-namespace
  kubectl create namespace my-namespace

گزینه‌ها

--allow-missing-template-keys     پیش‌فرض: true

اگر مقدار آن true باشد، هرگاه یک فیلد یا کلید نگاشت (map) در قالب وجود نداشته باشد، هر خطایی که در قالب‌ها رخ دهد نادیده گرفته می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath به کار می‌رود.
--dry-run string[="unchanged"]     پیش‌فرض: "none"

مقدار آن باید یکی از "none"، "server" یا "client" باشد. اگر راهبرد client انتخاب شود، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود و در واقع ارسالی صورت نمی‌گیرد. اگر راهبرد server انتخاب شود، درخواست به سمت سرور فرستاده می‌شود، اما منبع به‌طور ماندگار ذخیره نمی‌شود.
--field-manager string     پیش‌فرض: "kubectl-create"

نام مدیری که برای ردیابی مالکیت فیلدها به کار می‌رود.
-h, --help

راهنمای دستور namespace
-o, --output string

قالب خروجی. مقدار آن یکی از این موارد است: (json، yaml، kyaml، name، go-template، go-template-file، template، templatefile، jsonpath، jsonpath-as-json، jsonpath-file).
--save-config

اگر مقدار آن true باشد، پیکربندی آبجکت فعلی در حاشیه‌نویسی (annotation) آن ذخیره می‌شود؛ در غیر این صورت، حاشیه‌نویسی بدون تغییر باقی می‌ماند. این فلگ زمانی مفید است که بخواهید در آینده دستور kubectl apply را روی این آبجکت اجرا کنید.
--show-managed-fields

اگر مقدار آن true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields حفظ می‌شود.
--template string

رشته‌ی قالب یا مسیر فایل قالبی که هنگام استفاده از ‎-o=go-template یا ‎-o=go-template-file به کار می‌رود. قالب آن از نوع قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].
--validate string[="strict"]     پیش‌فرض: "strict"

مقدار آن باید یکی از این موارد باشد: strict (یا true)، warn، ignore (یا false). مقدار "true" یا "strict" ورودی را با استفاده از یک شِما اعتبارسنجی می‌کند و اگر ورودی نامعتبر باشد، درخواست با شکست مواجه می‌شود. اگر ServerSideFieldValidation روی api-server فعال باشد، اعتبارسنجی در سمت سرور انجام می‌شود؛ در غیر این صورت، به اعتبارسنجی سمت کلاینت که قابلیت اطمینان کمتری دارد بازمی‌گردد. اگر اعتبارسنجی فیلد در سمت سرور روی سرور API فعال باشد، مقدار "warn" درباره‌ی فیلدهای ناشناخته یا تکراری هشدار می‌دهد، بی‌آنکه جلوی درخواست را بگیرد؛ در غیر این صورت، همانند "ignore" رفتار می‌کند. مقدار "false" یا "ignore" هیچ اعتبارسنجی شِمایی انجام نمی‌دهد و فیلدهای ناشناخته یا تکراری را بی‌صدا کنار می‌گذارد.

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری‌ای که عملیات به جای آن (impersonate) انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام باشد.
--as-group strings

گروهی که عملیات به جای آن انجام می‌شود. برای مشخص کردن چند گروه می‌توانید این فلگ را چند بار تکرار کنید.
--as-uid string

شناسه‌ی UID که عملیات به جای آن انجام می‌شود.
--as-user-extra strings

اطلاعات تکمیلی کاربری که عملیات به جای آن انجام می‌شود. برای مشخص کردن چند مقدار برای یک کلید می‌توانید این فلگ را چند بار تکرار کنید.
--cache-dir string     پیش‌فرض: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام کانتکستی (context) از kubeconfig که باید استفاده شود
--disable-compression

اگر مقدار آن true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌هایی که به سرور فرستاده می‌شوند غیرفعال می‌شود
--insecure-skip-tls-verify

اگر مقدار آن true باشد، اعتبار گواهی سرور بررسی نمی‌شود. با این کار، اتصال‌های HTTPS شما ناامن خواهند شد
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های CLI به کار می‌رود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات (preferences) به کار می‌رود. می‌توانید این قابلیت را با export کردن feature gate با مقدار KUBECTL_KUBERC=false یا خاموش کردن قابلیت با KUBERC=off غیرفعال کنید.
--match-server-version

الزام می‌کند که نسخه‌ی سرور با نسخه‌ی کلاینت مطابقت داشته باشد
-n, --namespace string

در صورت تعیین، دامنه‌ی فضای نام این درخواست CLI را مشخص می‌کند
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در سرور API
--profile string     پیش‌فرض: "none"

نام پروفایلی که باید ثبت شود. یکی از این موارد: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     پیش‌فرض: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی که برای درخواست‌ها به سرور API به کار می‌رود
--request-timeout string     پیش‌فرض: "0"

مدت زمانی که پیش از صرف‌نظر کردن از یک درخواست واحد به سرور باید منتظر ماند. مقدارهای غیرصفر باید واحد زمانی متناظر داشته باشند (برای مثال 1s، 2m، 3h). مقدار صفر به این معناست که برای درخواست‌ها مهلت زمانی (timeout) در نظر گرفته نمی‌شود.
-s, --server string

نشانی و پورت سرور API کوبرنتیز
--storage-driver-buffer-duration duration     پیش‌فرض: 1m0s

عملیات نوشتن در درایور ذخیره‌سازی به اندازه‌ی این مدت در بافر نگه داشته می‌شوند و سپس در قالب یک تراکنش واحد در بک‌اندهای غیر حافظه‌ای (non-memory) ثبت می‌شوند
--storage-driver-db string     پیش‌فرض: "cadvisor"

نام پایگاه داده
--storage-driver-host string     پیش‌فرض: "localhost:8086"

میزبان و پورت پایگاه داده به شکل host:port

گذرواژه‌ی پایگاه داده

از اتصال امن با پایگاه داده استفاده می‌کند

--token string

توکن حامل (bearer token) برای احراز هویت در سرور API

نام کاربری از kubeconfig که باید استفاده شود

گزینه‌های ‎--version و ‎--version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینه‌ی ‎--version=vX.Y.Z...‎ نسخه‌ی گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافت‌شده از سرور را به‌عنوان خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود
