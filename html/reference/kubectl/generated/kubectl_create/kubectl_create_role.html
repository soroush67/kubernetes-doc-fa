kubectl create role

ایجاد یک نقش (role) با یک قاعده‌ی واحد

خلاصه

این دستور یک نقش (role) با یک قاعده‌ی واحد ایجاد می‌کند.

kubectl create role NAME --verb=verb --resource=resource.group/subresource [--resource-name=resourcename] [--dry-run=server|client|none]

مثال‌ها

  # Create a role named "pod-reader" that allows user to perform "get", "watch" and "list" on pods
  kubectl create role pod-reader --verb=get --verb=list --verb=watch --resource=pods
  
  # Create a role named "pod-reader" with ResourceName specified
  kubectl create role pod-reader --verb=get --resource=pods --resource-name=readablepod --resource-name=anotherpod
  
  # Create a role named "foo" with API Group specified
  kubectl create role foo --verb=get,list,watch --resource=rs.apps
  
  # Create a role named "foo" with SubResource specified
  kubectl create role foo --verb=get,list,watch --resource=pods,pods/status

گزینه‌ها

--allow-missing-template-keys     پیش‌فرض: true

اگر مقدار آن true باشد، هر خطایی که در قالب‌ها به‌دلیل نبودن یک فیلد یا کلید نگاشت (map key) در قالب رخ دهد، نادیده گرفته می‌شود. این گزینه فقط برای قالب‌های خروجی golang و jsonpath اعمال می‌شود.
--dry-run string[="unchanged"]     پیش‌فرض: "none"

مقدار آن باید یکی از "none"، "server" یا "client" باشد. اگر راهبرد client انتخاب شود، فقط آبجکتی که قرار است ارسال شود چاپ می‌شود و هیچ چیزی ارسال نمی‌شود. اگر راهبرد server انتخاب شود، درخواست سمت سرور ارسال می‌شود، اما منبع به‌طور ماندگار ذخیره نمی‌شود.
--field-manager string     پیش‌فرض: "kubectl-create"

نام مدیری (manager) که برای ردیابی مالکیت فیلدها به کار می‌رود.
-h, --help

راهنمای دستور role
-o, --output string

قالب خروجی. مقدار آن یکی از این موارد است: (json، yaml، kyaml، name، go-template، go-template-file، template، templatefile، jsonpath، jsonpath-as-json، jsonpath-file).
--resource strings

منبعی که قاعده روی آن اعمال می‌شود
--resource-name strings

منبعی در فهرست مجاز (white list) که قاعده روی آن اعمال می‌شود. برای مشخص‌کردن چند مورد، این فلگ را تکرار کنید.
--save-config

اگر مقدار آن true باشد، پیکربندی آبجکت فعلی در حاشیه‌نویسی (annotation) آن ذخیره می‌شود؛ در غیر این صورت، حاشیه‌نویسی بدون تغییر باقی می‌ماند. این فلگ زمانی مفید است که بخواهید در آینده دستور kubectl apply را روی این آبجکت اجرا کنید.
--show-managed-fields

اگر مقدار آن true باشد، هنگام چاپ آبجکت‌ها در قالب JSON یا YAML، فیلد managedFields نیز حفظ می‌شود.
--template string

رشته‌ی قالب یا مسیر فایل قالبی که هنگام استفاده از -o=go-template یا -o=go-template-file به کار می‌رود. قالب آن از نوع قالب‌های golang است [http://golang.org/pkg/text/template/#pkg-overview].
--validate string[="strict"]     پیش‌فرض: "strict"

مقدار آن باید یکی از این موارد باشد: strict (یا true)، warn یا ignore (یا false). مقدار "true" یا "strict" ورودی را با یک طرح‌واره (schema) اعتبارسنجی می‌کند و اگر ورودی نامعتبر باشد، درخواست را رد می‌کند. اگر ServerSideFieldValidation روی api-server فعال باشد، اعتبارسنجی سمت سرور انجام می‌شود؛ در غیر این صورت، به اعتبارسنجی سمت کلاینت که اطمینان کمتری دارد بازمی‌گردد. اگر اعتبارسنجی فیلد سمت سرور روی API server فعال باشد، مقدار "warn" درباره‌ی فیلدهای ناشناخته یا تکراری هشدار می‌دهد، بی‌آنکه درخواست را مسدود کند؛ در غیر این صورت، مانند "ignore" رفتار می‌کند. مقدار "false" یا "ignore" هیچ اعتبارسنجی طرح‌واره‌ای انجام نمی‌دهد و هر فیلد ناشناخته یا تکراری را بی‌صدا کنار می‌گذارد.
--verb strings

فعلی (verb) که روی منابع موجود در قاعده اعمال می‌شود

گزینه‌های ارث‌بری شده از والد

--as string

نام کاربری که عملیات به جای او (impersonate) انجام می‌شود. این کاربر می‌تواند یک کاربر عادی یا یک حساب سرویس (service account) در یک فضای نام (namespace) باشد.
--as-group strings

گروهی که عملیات به جای آن انجام می‌شود. برای مشخص‌کردن چند گروه می‌توان این فلگ را تکرار کرد.
--as-uid string

شناسه‌ی UID که عملیات به جای آن انجام می‌شود.
--as-user-extra strings

اطلاعات اضافی کاربر که عملیات به جای آن انجام می‌شود. برای مشخص‌کردن چند مقدار برای یک کلید، می‌توان این فلگ را تکرار کرد.
--cache-dir string     پیش‌فرض: "$HOME/.kube/cache"

دایرکتوری پیش‌فرض حافظه‌ی نهان (cache)
--certificate-authority string

مسیر فایل گواهی مرجع صدور گواهی (certificate authority)
--client-certificate string

مسیر فایل گواهی کلاینت برای TLS
--client-key string

مسیر فایل کلید کلاینت برای TLS
--cluster string

نام کلاستری از kubeconfig که باید استفاده شود
--context string

نام زمینه‌ای (context) از kubeconfig که باید استفاده شود
--disable-compression

اگر مقدار آن true باشد، فشرده‌سازی پاسخ برای همه‌ی درخواست‌هایی که به سرور ارسال می‌شوند غیرفعال می‌شود
--insecure-skip-tls-verify

اگر مقدار آن true باشد، اعتبار گواهی سرور بررسی نمی‌شود. با این کار، اتصال‌های HTTPS شما ناامن می‌شوند
--kubeconfig string

مسیر فایل kubeconfig که برای درخواست‌های CLI استفاده می‌شود.
--kuberc string

مسیر فایل kuberc که برای ترجیحات (preferences) استفاده می‌شود. می‌توان این قابلیت را با صادرکردن (export) متغیر feature gate به‌صورت KUBECTL_KUBERC=false یا با خاموش‌کردن قابلیت به‌صورت KUBERC=off غیرفعال کرد.
--match-server-version

الزام می‌کند که نسخه‌ی سرور با نسخه‌ی کلاینت یکسان باشد
-n, --namespace string

در صورت تعیین، دامنه‌ی فضای نام برای این درخواست CLI را مشخص می‌کند
--password string

گذرواژه برای احراز هویت پایه (basic authentication) در API server
--profile string     پیش‌فرض: "none"

نام پروفایلی که باید ثبت شود. مقدار آن یکی از این موارد است: (none|cpu|heap|goroutine|threadcreate|block|mutex|trace)
--profile-output string     پیش‌فرض: "profile.pprof"

نام فایلی که پروفایل در آن نوشته می‌شود
--proxy-url string

نشانی URL پراکسی که برای درخواست‌های ارسالی به API server استفاده می‌شود
--request-timeout string     پیش‌فرض: "0"

مدت زمانی که پیش از رهاکردن یک درخواست واحد به سرور باید منتظر ماند. مقدارهای غیرصفر باید واحد زمانی متناظر داشته باشند (برای مثال 1s، 2m، 3h). مقدار صفر یعنی برای درخواست‌ها هیچ مهلت زمانی (timeout) در نظر گرفته نمی‌شود.
-s, --server string

نشانی و درگاه (port) سرور API کوبرنتیز
--storage-driver-buffer-duration duration     پیش‌فرض: 1m0s

نوشتن‌ها در درایور ذخیره‌سازی به‌مدت این بازه‌ی زمانی در بافر نگه داشته می‌شوند و سپس به‌صورت یک تراکنش واحد در پشتیبان‌های غیرحافظه‌ای (non memory backends) ثبت می‌شوند
--storage-driver-db string     پیش‌فرض: "cadvisor"

نام پایگاه داده
--storage-driver-host string     پیش‌فرض: "localhost:8086"

میزبان و درگاه پایگاه داده به‌صورت host:port
--storage-driver-password string     پیش‌فرض: "root"

گذرواژه‌ی پایگاه داده
--storage-driver-secure

از اتصال امن با پایگاه داده استفاده می‌کند
--storage-driver-table string     پیش‌فرض: "stats"

نام جدول
--storage-driver-user string     پیش‌فرض: "root"

نام کاربری پایگاه داده
--tls-server-name string

نام سروری که برای اعتبارسنجی گواهی سرور استفاده می‌شود. اگر این مقدار ارائه نشود، از نام میزبانی (hostname) استفاده می‌شود که برای برقراری ارتباط با سرور به کار رفته است
--token string

توکن حامل (bearer token) برای احراز هویت در API server
--user string

نام کاربری از kubeconfig که باید استفاده شود
--username string

نام کاربری برای احراز هویت پایه در API server
--version version[=true]

گزینه‌های --version و --version=raw اطلاعات نسخه را چاپ می‌کنند و از برنامه خارج می‌شوند؛ گزینه‌ی --version=vX.Y.Z... نسخه‌ی گزارش‌شده را تنظیم می‌کند
--warnings-as-errors

هشدارهای دریافتی از سرور را به‌عنوان خطا در نظر می‌گیرد و با کد خروج غیرصفر خارج می‌شود
